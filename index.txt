let isRegisterMode = false;

function toggleAuthMode() {
    isRegisterMode = !isRegisterMode;
    const btn = document.getElementById('auth-btn');
    const toggle = document.getElementById('auth-toggle');
    
    if (isRegisterMode) {
        btn.innerHTML = `<i class="fa-solid fa-user-plus"></i> Daftar`;

async function handleLogin() {
    const username = document.getElementById('auth-username').value.trim();
    const password = document.getElementById('auth-password').value.trim();
    const btn = document.getElementById('auth-btn');
    const msg = document.getElementById('auth-message');
    
    if (!username || !password) {
        msg.textContent = '❌ Isi username dan password!';
        msg.className = 'text-center text-xs mt-3 text-red-400';
        msg.classList.remove('hidden');
        return;
    }
    
    btn.disabled = true;
    btn.innerHTML = `<i class="fa-solid fa-circle-notch fa-spin"></i> Loading...`;
    
    try {
        if (isRegisterMode) {
            // ===== REGISTER =====
            const hashedPassword = await bcrypt.hash(password, 10);
            
            const { data, error } = await supabaseClient
                .from('profiles')
                .insert([{ username, password: hashedPassword, role: 'user' }])
                .select();
            
            if (error) throw error;
            
            msg.textContent = '✅ Berhasil daftar! Silakan login.';
            msg.className = 'text-center text-xs mt-3 text-green-400';
            msg.classList.remove('hidden');
            setTimeout(() => toggleAuthMode(), 1500);
            
        } else {
            // ===== LOGIN =====
            const { data, error } = await supabaseClient
                .from('profiles')
                .select('*')
                .eq('username', username)
                .single();
            
            if (error || !data) throw new Error('Username tidak ditemukan!');
            
            const passwordMatch = await bcrypt.compare(password, data.password);
            
            if (!passwordMatch) throw new Error('Password salah!');
            
            localStorage.setItem('user_id', data.id);
            localStorage.setItem('username', data.username);
            localStorage.setItem('role', data.role);
            
            msg.textContent = '✅ Login berhasil!';
            msg.className = 'text-center text-xs mt-3 text-green-400';
            msg.classList.remove('hidden');
            
            setTimeout(() => {
                document.getElementById('auth-page').classList.add('hidden');
                loadUserProfile();
            }, 1000);
        }
    } catch (err) {
        console.error(err);
        msg.textContent = '❌ ' + (err.message || 'Terjadi kesalahan');
        msg.className = 'text-center text-xs mt-3 text-red-400';
        msg.classList.remove('hidden');
    } finally {
        btn.disabled = false;
        if (isRegisterMode) {
            btn.innerHTML = `<i class="fa-solid fa-user-plus"></i> Daftar`;
        } else {
            btn.innerHTML = `<i class="fa-solid fa-right-to-bracket"></i> Login`;
        }
    }
}

function loadUserProfile() {
    const username = localStorage.getItem('username') || 'Guest';
    const role = localStorage.getItem('role') || 'user';
    
    const nameEl = document.querySelector('h2.text-lg.font-black');
    if (nameEl) nameEl.textContent = username;
    
    const roleBadge = document.getElementById('user-role-badge');
    if (roleBadge) {
        const roles = {
            admin: { label: 'Admin', color: 'bg-red-500/20 text-red-400 border-red-500/40', icon: 'fa-crown' },
            moderator: { label: 'Mod', color: 'bg-orange-500/20 text-orange-400 border-orange-500/40', icon: 'fa-shield' },

            user: { label: 'User', color: 'bg-blue-500/20 text-blue-400 border-blue-500/40', icon: 'fa-user' }

        roleBadge.innerHTML = `<i class="fa-solid ${r.icon} text-[9px]"></i> ${r.label}`;
    }
